class BoundingBoxDrawer {
    clearAll() {
        // Remove all existing bounding boxes from canvas
        this.boxes = [];
        this.redrawCanvas();
    }
    
    drawFromMasks(masks) {
        // Step 1: Clear everything
        this.clearAll();
        
        // Step 2: Draw only the masks in the array
        masks.forEach(mask => {
            const [x, y, w, h] = mask.bounding_box;
            this.drawBox(x, y, w, h, '#00ff00');  // Green box
            this.boxes.push({id: mask.mask_id, rect: [x, y, w, h]});
        });
    }
}

// Usage pattern for ALL endpoints:
function handleBackendResponse(response) {
    if (response.success && response.masks !== undefined) {
        boxDrawer.drawFromMasks(response.masks);
        // This works for:
        // - Upload (masks: [] → clears all)
        // - Segmentation (masks: [...45 items] → draws 45 boxes)
        // - Intensity filter (masks: [...20 items] → draws 20 boxes)
        // - Reset filter (masks: [...45 items] → draws 45 boxes)
    }
}
